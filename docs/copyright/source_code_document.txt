def _auth_singleton():
    """The earthaccess Auth instance, tolerating API changes across versions.

    earthaccess 0.19 keeps the live instance in the package-private
    `earthaccess.__auth__`; older releases exposed it as `earthaccess.auth`.
    In 0.19 `earthaccess.auth` is a MODULE, so non-instances are skipped.
    """
    import types

    for name in ("__auth__", "auth"):
        try:
            candidate = getattr(earthaccess, name)
        except AttributeError:
            continue
        if candidate is None:
            continue
        if isinstance(candidate, (type, types.ModuleType)):
            continue
        if hasattr(candidate, "authenticated"):
            return candidate
    return None

def try_restore_session() -> bool:
    """Silently restore a persisted login from ~/.netrc (best effort).

    earthaccess's netrc strategy exchanges the stored credentials for a fresh
    URS token, so this needs network; failures are silent.
    """
    if earthaccess is None:
        return False
    if is_authenticated():
        return True
    try:
        _call_login(strategy="netrc", persist=False)
    except Exception:
        return False
    return is_authenticated()

def _urs_probe(username: str, password: str) -> str:
    """Ask URS directly what it thinks of the credential pair.

    Returns one of: "ok", "bad-credentials", "blocked", "network", "status-<n>".
    """
    import base64

    import requests

    try:
        pair = base64.b64encode(f"{username}:{password}".encode()).decode()
        resp = requests.post(
            "https://urs.earthdata.nasa.gov/api/users/token",
            headers={"Authorization": f"Basic {pair}"},
            timeout=30,
        )
    except requests.RequestException:
        return "network"
    if resp.status_code == 200:
        return "ok"
    if resp.status_code == 401:
        return "bad-credentials"
    if resp.status_code in (403, 429):
        return "blocked"
    return f"status-{resp.status_code}"

def _cmd_download(args):
    if not auth.is_authenticated():
        if sys.stdin.isatty():
            print("not signed in; downloading requires Earthdata login")
            return _cmd_login(args)
        print("not signed in (no tty). Run `d3ltool login` first.", file=sys.stderr)
        return 2

    params = _params_from_args(args)
    granules = search.run_search(params)
    if not granules:
        print("no granules found")
        return 0

    cfg = config.load_config()
    dest = args.dest or cfg["download_dir"]
    workers = args.workers or int(cfg.get("max_workers", 3))

    def on_event(item, event, payload):
        if item is None:
            return
        if event == "status" and item.state in ("done", "skipped", "failed", "canceled"):
            mark = {"done": "OK ", "skipped": "SKIP", "failed": "FAIL",
                    "canceled": "CANC"}.get(item.state, "    ")
            print(f"[{mark}] {item.filename} {item.message}")
        elif event == "progress" and item.total:
            sys.stdout.write(f"\r  {item.filename}: {item.downloaded / 1048576:.1f}/"
                             f"{item.total / 1048576:.1f} MB  "
                             f"{item.speed / 1048576:.1f} MB/s   ")
            sys.stdout.flush()

    eng = engine.DownloadEngine(session_factory=auth.new_download_session,
                                dest_dir=dest, max_workers=workers,
                                on_event=on_event)
    print(f"downloading {len(granules)} granule(s) -> {dest} (workers={workers})")
    for g in granules:
        links = search.granule_links(g)
        if not links:
            print(f"[FAIL] {search.granule_filename(g)}: no download link")
            continue
        size_mb = search.granule_size_mb(g)
        eng.add(links[0], filename=search.granule_filename(g),
                size_hint=int(size_mb * 1048576) if size_mb else 0)
    eng.wait_all()
    print("done.")
    return 0

    )

    cfg = load_config()
    cfg.update({k: v for k, v in kw.items() if k in DEFAULTS})
    save_config(cfg)
    return cfg

def tiles_for_bbox(west: float, south: float, east: float, north: float) -> list:
    """Exact bbox -> [(h, v)] on the sinusoidal grid — the Python twin of the
    R package's glass_tiles_for_bbox().  Tiles are exact rectangles in
    sinusoidal space, so the bbox intersects a tile iff their latitude bands
    overlap and the bbox longitude interval overlaps the tile's longitude
    interval at some latitude inside that overlap.  Both inequalities are

====================================================================================================
    monotone in |lat| (a tile's edges move outward as cos(lat) shrinks), so
    evaluating each at its analytically hardest latitude inside the overlap
    is exact — no polygon library needed.  Rows reaching into the polar caps
    (|lat| >= 80, where a tile's longitude interval exceeds the row's own
    width and wraps the antimeridian) degrade to whole-row inclusion:
    over-inclusion merely lists tiles with no matching files, under-inclusion
    loses data."""
    west, south, east, north = (float(x) for x in (west, south, east, north))
    if west > east or south > north:
        return []
    out = []
    for v in range(_tiles.N_ROWS):
        lat_max = 90.0 - 10.0 * v
        lat_min = lat_max - 10.0
        lo, hi = max(south, lat_min), min(north, lat_max)
        if lo > hi:
            continue
        if max(abs(lo), abs(hi)) >= 80.0:
            out.extend((h, v) for h in range(_tiles.N_COLS))
            continue
        # |lat| extremes within the overlapping band
        if lo <= 0.0 <= hi:
            m = 0.0
        else:
            m = lo if abs(lo) < abs(hi) else hi
        M = lo if abs(lo) >= abs(hi) else hi
        c_m = max(math.cos(math.radians(m)), 1e-9)
        c_M = max(math.cos(math.radians(M)), 1e-9)
        for h in range(_tiles.N_COLS):
            x0 = (h - 18) * _tiles.TILE_SIZE
            x1 = x0 + _tiles.TILE_SIZE
            if x1 <= 0.0:      # west of the 0 meridian: edges move west with |lat|
                right = math.degrees(x1 / (_tiles.R_SIN * c_M))
                left = math.degrees(x0 / (_tiles.R_SIN * c_m))
            elif x0 >= 0.0:    # east of the 0 meridian: edges move east with |lat|
                right = math.degrees(x1 / (_tiles.R_SIN * c_m))
                left = math.degrees(x0 / (_tiles.R_SIN * c_M))
            else:              # straddles the 0 meridian (h == 17)
                right = math.degrees(x1 / (_tiles.R_SIN * c_m))
                left = math.degrees(x0 / (_tiles.R_SIN * c_m))
            if west <= right and left <= east:
                out.append((h, v))
    return out

def _fetch_hrefs(url: str, session) -> list:
    """All <a href> values of a directory page, with the glass.hku.hk
    politeness policy: consecutive directory fetches are spaced at least
    LISTING_MIN_INTERVAL apart, and transient failures (timeouts, HTTP
    429/5xx — the server's way of rate-limiting) retry on the measured
    exponential backoff 8/24/72s plus one final attempt after a 15s
    cooldown.  404 is never retried: it becomes FileNotFoundError, a normal
    sparse-folder condition for the search loop."""
    waits = (0.0,) + tuple(LISTING_BACKOFFS) + (LISTING_COOLDOWN,)
    last_exc = None
    for wait in waits:
        if wait:
            time.sleep(wait)
        gap = LISTING_MIN_INTERVAL - (time.monotonic() - _LAST_LISTING[0])
        if gap > 0:
