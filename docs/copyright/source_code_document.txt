def _auth_singleton():
    """The earthaccess Auth instance, tolerating API changes across versions.

    earthaccess 0.19 keeps the live instance in the package-private
    `earthaccess.__auth__`; older releases exposed it as `earthaccess.auth`.
    In 0.19 `earthaccess.auth` is a MODULE, so non-instances are skipped.
    """
    import types

    for name in ("__auth__", "auth"):
        try:
            candidate = getattr(earthaccess, name)
        except AttributeError:
            continue
        if candidate is None:
            continue
        if isinstance(candidate, (type, types.ModuleType)):
            continue
        if hasattr(candidate, "authenticated"):
            return candidate
    return None

def try_restore_session() -> bool:
    """Silently restore a persisted login from ~/.netrc (best effort).

    earthaccess's netrc strategy exchanges the stored credentials for a fresh
    URS token, so this needs network; failures are silent.
    """
    if earthaccess is None:
        return False
    if is_authenticated():
        return True
    try:
        _call_login(strategy="netrc", persist=False)
    except Exception:
        return False
    return is_authenticated()

def _urs_probe(username: str, password: str) -> str:
    """Ask URS directly what it thinks of the credential pair.

    Returns one of: "ok", "bad-credentials", "blocked", "network", "status-<n>".
    """
    import base64

    import requests

    try:
        pair = base64.b64encode(f"{username}:{password}".encode()).decode()
        resp = requests.post(
            "https://urs.earthdata.nasa.gov/api/users/token",
            headers={"Authorization": f"Basic {pair}"},
            timeout=30,
        )
    except requests.RequestException:
        return "network"
    if resp.status_code == 200:
        return "ok"
    if resp.status_code == 401:
        return "bad-credentials"
    if resp.status_code in (403, 429):
        return "blocked"
    return f"status-{resp.status_code}"

def _cmd_download(args):
    if not auth.is_authenticated():
        if sys.stdin.isatty():
            print("not signed in; downloading requires Earthdata login")
            return _cmd_login(args)
        print("not signed in (no tty). Run `d3ltool login` first.", file=sys.stderr)
        return 2

    params = _params_from_args(args)
    granules = search.run_search(params)
    if not granules:
        print("no granules found")
        return 0

    cfg = config.load_config()
    dest = args.dest or cfg["download_dir"]
    workers = args.workers or int(cfg.get("max_workers", 3))

    def on_event(item, event, payload):
        if item is None:
            return
        if event == "status" and item.state in ("done", "skipped", "failed", "canceled"):
            mark = {"done": "OK ", "skipped": "SKIP", "failed": "FAIL",
                    "canceled": "CANC"}.get(item.state, "    ")
            print(f"[{mark}] {item.filename} {item.message}")
        elif event == "progress" and item.total:
            sys.stdout.write(f"\r  {item.filename}: {item.downloaded / 1048576:.1f}/"
                             f"{item.total / 1048576:.1f} MB  "
                             f"{item.speed / 1048576:.1f} MB/s   ")
            sys.stdout.flush()

    eng = engine.DownloadEngine(session_factory=auth.new_download_session,
                                dest_dir=dest, max_workers=workers,
                                on_event=on_event)
    print(f"downloading {len(granules)} granule(s) -> {dest} (workers={workers})")
    for g in granules:
        links = search.granule_links(g)
        if not links:
            print(f"[FAIL] {search.granule_filename(g)}: no download link")
            continue
        size_mb = search.granule_size_mb(g)
        eng.add(links[0], filename=search.granule_filename(g),
                size_hint=int(size_mb * 1048576) if size_mb else 0)
    eng.wait_all()
    print("done.")
    return 0

    cfg.update({k: v for k, v in kw.items() if k in DEFAULTS})
    save_config(cfg)
    return cfg

def doys(v: dict, start: str, end: str, mode: str = "span") -> list:
    """Port of the R glass_doys(): expand a date range into composite
    (year, doy) folder keys with the variant's cadence.  "span" (default)
    keeps every composite whose interval [begin, begin+cadence-1] intersects
    the range; "start" keeps composites beginning inside the range.  Folder
    start DOYs cap at 365 like the R original (leap years keep folder 365).
    Unknown cadence raises GlassError — search() falls back to a
    listing-derived expansion for those skeleton variants."""
    cad = v.get("cadence_days")
    if not cad:
        raise GlassError(f"cadence unknown for variant {v['variant_id']}; "
                         "composite folders are derived from the archive "
                         "instead")
    d0, d1 = _date_range(start, end)
    out = []
    for year in range(d0.year, d1.year + 1):
        for doy in range(1, 366, cad):
            begin = date(year, 1, 1) + timedelta(days=doy - 1)
            last = begin + timedelta(days=cad - 1)
            if (begin <= d1 and last >= d0) if mode == "span" \
                    else (d0 <= begin <= d1):
                out.append((year, doy))
    return out

def tiles_for_bbox(west: float, south: float, east: float, north: float) -> list:

    sinusoidal space, so the bbox intersects a tile iff their latitude bands
    overlap and the bbox longitude interval overlaps the tile's longitude
    interval at some latitude inside that overlap.  Both inequalities are
    monotone in |lat| (a tile's edges move outward as cos(lat) shrinks), so
    evaluating each at its analytically hardest latitude inside the overlap
    is exact — no polygon library needed.  Rows reaching into the polar caps
    (|lat| >= 80, where a tile's longitude interval exceeds the row's own
    width and wraps the antimeridian) degrade to whole-row inclusion:
    over-inclusion merely lists tiles with no matching files, under-inclusion
    loses data."""

    if west > east or south > north:
        return []
    out = []
    for v in range(_tiles.N_ROWS):
        lat_max = 90.0 - 10.0 * v
        lat_min = lat_max - 10.0
        lo, hi = max(south, lat_min), min(north, lat_max)

            continue

        c_M = max(math.cos(math.radians(M)), 1e-9)

class _ThreadAdapter:
    """Bridges data-layer progress calls to the thread's Qt signals."""

    def __init__(self, thread):
        self._thread = thread

    def stage(self, key, **kw):
        self._thread.stage.emit(i18n.tr(key, **kw))

    def layers(self, layers):
        self._thread.layers_ready.emit(layers)


class TimeSeriesThread(QThread):
    """The LAADS OPeNDAP pipeline (search -> probe -> sample) off the GUI
    thread; one job at a time, stoppable between granules."""

    layers_ready = Signal(list)          # [{layer, description, ...}]
    series_ready = Signal(list)          # [source_label, layer, points]
    stage = Signal(str)                  # human-readable progress text

    failed = Signal(str)
    finished_turn = Signal()

    def __init__(self, args, parent=None):
        super().__init__(parent)
        self._args = args
        self._stop_event = threading.Event()

    def stop(self):
        self._stop_event.set()

    def run(self):
        try:
            source, layer, points, layer_meta = tsl.run_point_series(
                lat=self._args["lat"], lon=self._args["lon"],
                product=self._args["product"],
                start=self._args["start"], end=self._args["end"],
                layer_hint=self._args.get("layer_hint", ""),
                should_stop=self._stop_event.is_set,
                adapter=_ThreadAdapter(self))
            if self._stop_event.is_set():
                raise tsl.StopRequested("stopped by user")
            self.series_ready.emit([source, layer, points, layer_meta])
        except tsl.StopRequested:
            pass
        except Exception as exc:      # noqa: BLE001 — the panel must not die
            self.failed.emit(f"{type(exc).__name__}: {exc}")
        finally:
            self.finished_turn.emit()

    def _plot(self):
        if self._thread is not None and self._thread.isRunning():
            return
        try:
            lat = float(self.lat_edit.text().strip())
            lon = float(self.lon_edit.text().strip())
            if not (-90.0 <= lat <= 90.0 and -180.0 <= lon <= 180.0):
                raise ValueError("lat/lon out of range")
        except ValueError:
            self._status(i18n.tr("ts_bad_coords"), error=True)
            return
        start = self.start_edit.date().toString("yyyy-MM-dd")
        end = self.end_edit.date().toString("yyyy-MM-dd")
        if start > end:
            self._status(i18n.tr("ts_bad_dates"), error=True)
            return
        product = self._typed_product()
        if not product:
            self._status(i18n.tr("ts_need_fields",
                                 fields=i18n.tr("ts_field_product")),
                         error=True)
            return
        layer_hint = self.layer_combo.currentData() or \
            self.layer_combo.currentText()

        self.plot_btn.setEnabled(False)
        self.stop_btn.setEnabled(True)
        self._status(i18n.tr("ts_stage_search"))
        self._thread = TimeSeriesThread({
            "lat": lat, "lon": lon, "product": product,
            "layer_hint": layer_hint, "start": start, "end": end,
        }, parent=self)
        self._thread.layers_ready.connect(self._apply_layers)
        self._thread.series_ready.connect(self._on_series_ready)
        self._thread.stage.connect(self._status)
        self._thread.failed.connect(self._on_failed)
        self._thread.finished_turn.connect(self._on_finished)
        self._thread.start()

    def _apply_layers(self, layers):
        current = self.layer_combo.currentData()
        self.layer_combo.blockSignals(True)
        self.layer_combo.clear()
        for entry in layers:
            name = str(entry.get("layer", ""))
            if not name:
                continue
            desc = str(entry.get("description", "") or "")
            label = f"{name} — {desc}" if desc else name
            self.layer_combo.addItem(label, name)
        self.layer_combo.blockSignals(False)
        if current:
            index = self.layer_combo.findData(current)
            if index >= 0:
                self.layer_combo.setCurrentIndex(index)

    def render_points(self, points):
        """Draw (or clear with []) the series; records self.last_points."""
        self.chart.removeAllSeries()
        for axis in self.chart.axes():    # removeAllSeries keeps old axes,
            self.chart.removeAxis(axis)   # so labels pile up on every re-plot
        self.last_points = [list(p) for p in (points or [])]
        if not points:
            return
        series = QLineSeries()
        times = []
        for iso, value in points:
            text = str(iso)
            moment = QDateTime.fromString(text, Qt.ISODate)
            if not moment.isValid():      # "2025-10-25T01:30" parses directly;
                moment = QDateTime.fromString(   # pad date-only stamps
                    text[:10] + "T00:00:00", Qt.ISODate)
            if not moment.isValid():
                continue
            times.append(moment)
            series.append(moment.toMSecsSinceEpoch(), float(value))
        if not times:
            return
        pen = QPen(QColor("#d6372a"))
        pen.setWidthF(1.8)
        series.setPen(pen)

        x_axis = QDateTimeAxis()
        span_days = times[0].secsTo(times[-1]) // 86400
        x_axis.setFormat("yyyy-MM-dd" if span_days <= 120 else "yyyy-MM")
        x_axis.setRange(times[0], times[-1])
        values = [float(v) for _, v in points]
        lo, hi = min(values), max(values)
        if lo == hi:
            lo, hi = lo - 1.0, hi + 1.0

"""PyInstaller entry script for the D3LToolNASA.exe build."""

import os
import sys

# windowed builds have no stdout/stderr; give them a sink so stray prints
# (ours, or warnings from libraries) never crash the app
if sys.stdout is None:
    sys.stdout = open(os.devnull, "w", encoding="utf-8")
if sys.stderr is None:
    sys.stderr = open(os.devnull, "w", encoding="utf-8")


def smoke() -> int:
    """Build-time verification: construct the whole UI headlessly, then exit."""
    os.environ.setdefault("QT_QPA_PLATFORM", "offscreen")
    from PySide6.QtWidgets import QApplication

    from d3ltool.gui.main_window import MainWindow
    from d3ltool.gui.theme import apply_theme

    app = QApplication([])
    apply_theme(app)
    window = MainWindow()
    window.close()
    try:
        print("SMOKE OK")
    except Exception:
        pass
    return 0


if __name__ == "__main__":
    if "--smoke" in sys.argv:
        sys.exit(smoke())
    from d3ltool.app import run_gui

    sys.exit(run_gui())
